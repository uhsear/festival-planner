## Font Families

export function FontFamilyDemo({ name, family }) {
  return (
    <div style={{ marginBottom: 24 }}>
      <div style={{ fontFamily: 'monospace', fontSize: 12, color: '#9999bb', marginBottom: 4 }}>
        fontFamily.{name}
      </div>
      <div style={{ fontFamily: family, fontSize: 24, color: '#eaeaf2', marginBottom: 4 }}>
        The quick brown fox jumps over the lazy dog
      </div>
      <div style={{ fontFamily: 'monospace', fontSize: 11, color: '#8787a8' }}>
        {family}
      </div>
    </div>
  );
}

## Font Size Ramp

Each size shown with rendered text at that size.

export function FontSizeDemo({ size, px }) {
  return (
    <div style={{
      display: 'flex',
      alignItems: 'baseline',
      gap: 16,
      marginBottom: 12,
      paddingBottom: 12,
      borderBottom: '1px solid rgba(255,255,255,0.04)',
    }}>
      <div style={{
        fontFamily: 'monospace',
        fontSize: 12,
        color: '#9999bb',
        width: 100,
        flexShrink: 0,
        textAlign: 'right',
      }}>
        fontSize.{size}
      </div>
      <div style={{
        fontFamily: 'monospace',
        fontSize: 12,
        color: '#ffb020',
        width: 50,
        flexShrink: 0,
      }}>
        {px}px
      </div>
      <div style={{
        fontFamily: "'Space Grotesk', system-ui, sans-serif",
        fontSize: px,
        color: '#eaeaf2',
        lineHeight: 1.3,
      }}>
        Festival Schedule
      </div>
    </div>
  );
}

{Object.entries(fontSize).map(([key, value]) => (
  <FontSizeDemo key={key} size={key} px={value} />
))}

## Letter Spacing

Values in `em` units (multiply by font size for px).

export function LetterSpacingDemo({ name, value }) {
  const label = value > 0 ? `+${value}em` : `${value}em`;
  return (
    <div style={{
      display: 'flex',
      alignItems: 'center',
      gap: 16,
      marginBottom: 12,
      paddingBottom: 12,
      borderBottom: '1px solid rgba(255,255,255,0.04)',
    }}>
      <div style={{
        fontFamily: 'monospace',
        fontSize: 12,
        color: '#9999bb',
        width: 140,
        flexShrink: 0,
      }}>
        letterSpacing.{name}
      </div>
      <div style={{
        fontFamily: 'monospace',
        fontSize: 12,
        color: '#ffb020',
        width: 70,
        flexShrink: 0,
      }}>
        {label}
      </div>
      <div style={{
        fontFamily: name === 'display'
          ? "'Syncopate', sans-serif"
          : "'Space Grotesk', system-ui, sans-serif",
        fontSize: name === 'display' ? 20 : 16,
        letterSpacing: `${value}em`,
        color: '#eaeaf2',
        textTransform: name === 'caps' ? 'uppercase' : 'none',
      }}>
        {name === 'display' ? 'FESTIVAL 2026' : name === 'caps' ? 'CREW NAME' : 'Body text example'}
      </div>
    </div>
  );
}

{Object.entries(letterSpacing).map(([key, value]) => (
  <LetterSpacingDemo key={key} name={key} value={value} />
))}

## Usage Reference

| Token | CSS Custom Property | Usage |
|-------|-------------------|-------|
| `fontFamily.body` | `--font-body` | All body text, UI labels |
| `fontFamily.display` | `--font-display` | Headlines, festival names |
| `fontSize.10` | `--text-10` | Tiny labels, badges |
| `fontSize.12` | `--text-12` | Captions, metadata |
| `fontSize.14` | `--text-14` | Default body text |
| `fontSize.16` | `--text-16` | Emphasized body |
| `fontSize.18` | `--text-18` | Subheadings |
| `fontSize.20` | `--text-20` | Section titles |
| `fontSize.24` | `--text-24` | Page headings |
| `fontSize.32` | `--text-32` | Hero text |
| `fontSize.48` | `--text-48` | Display / splash |
